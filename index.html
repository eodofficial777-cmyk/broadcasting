/** * 廣播室 · 成就與通關回報接收端 * * 用法： * 1. 開一份新的 Google 試算表 → 擴充功能 → Apps Script * 2. 把這整份貼進去，存檔 * 3. 部署 → 新增部署作業 → 類型選「網頁應用程式」 * 執行身分：我　　　有權存取的使用者：任何人 * 4. 複製產生的網址，貼回 HTML 的 REPORT： * const REPORT = { * mode: 'appsscript', * url: '貼在這裡', * ... * }; * 5. 想收通關通知就把下面的 NOTIFY_EMAIL 填上你的信箱 * * 玩家沒網路時回報會先存在瀏覽器裡，下次開啟自動補送，不會漏。 */ const SHEET_NAME = '回報'; const NOTIFY_EMAIL = ''; // 留空就不寄信 const DEDUP = true; // 同一個角色的同一個成就只記一次 function doPost(e) { const lock = LockService.getScriptLock(); try { lock.waitLock(20000); const d = JSON.parse(e.postData.contents); const sh = getSheet_(); const key = [d.charId || d.role, d.event, d.id || d.name].join('|'); if (DEDUP && seen_(sh, key)) { return json_({ ok: true, skipped: 'duplicate' }); } sh.appendRow([ new Date(), d.role || '', d.charId || '', d.faction || '', d.race || '', d.event || '', d.id || '', d.name || '', d.detail || '', d.time || '', key, ]); if (NOTIFY_EMAIL && d.event === 'clear') { MailApp.sendEmail( NOTIFY_EMAIL, '【廣播室】' + (d.role || '未登記') + ' 通關了', [ '角色：' + (d.role || ''), '陣營：' + (d.faction || '') + '　種族：' + (d.race || ''), '成就：' + (d.name || ''), '進度：' + (d.detail || ''), '時間：' + (d.time || ''), ].join('\n') ); } return json_({ ok: true }); } catch (err) { return json_({ ok: false, err: String(err) }); } finally { lock.releaseLock(); } } function doGet() { return ContentService.createTextOutput('ok'); } /* ── 內部用 ── */ function getSheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sh = ss.getSheetByName(SHEET_NAME); if (!sh) { sh = ss.insertSheet(SHEET_NAME); sh.appendRow(['收到時間', '角色', 'characterId', '陣營', '種族', '事件', '成就ID', '成就', '進度', '遊戲內時間', 'key']); sh.setFrozenRows(1); sh.hideColumns(11); } return sh; } function seen_(sh, key) { const last = sh.getLastRow(); if (last < 2) return false; const keys = sh.getRange(2, 11, last - 1, 1).getValues(); for (let i = 0; i < keys.length; i++) { if (keys[i][0] === key) return true; } return false; } function json_(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } /** * 選用：在試算表上跑一次，做一張統計表 * （每個成就幾個人拿到、通關人數） */ function 產生統計() { const sh = getSheet_(); const rows = sh.getDataRange().getValues().slice(1); const count = {}; const cleared = new Set(); rows.forEach(r => { const name = r[7] || '(無)'; count[name] = (count[name] || 0) + 1; if (r[5] === 'clear') cleared.add(r[2] || r[1]); if (r[5] === 'test') return; }); const ss = SpreadsheetApp.getActiveSpreadsheet(); let st = ss.getSheetByName('統計'); if (!st) st = ss.insertSheet('統計'); st.clear(); st.appendRow(['成就', '達成人次']); Object.keys(count).sort((a, b) => count[b] - count[a]) .forEach(k => st.appendRow([k, count[k]])); st.appendRow([]); st.appendRow(['通關人數', cleared.size]); }
